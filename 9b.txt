<script>
function factorial(num){
    if(num==0 || num==1)
        return 1;
    else
        return num * factorial(num-1);
}
function calculateFactorial(){
    let num = parseInt(document.getElementById("inputNum").value);
    document.getElementById("output").innerHTML =
    "Factorial: " + factorial(num);
}
function fibonacci(num){
    let fib = [0,1];
    for(let i=2;i<num;i++){
        fib[i] = fib[i-1] + fib[i-2];
    }
    return fib.slice(0,num).join(", ");
}
function displayFibonacci(){
    let num = parseInt(document.getElementById("inputNum").value);
    document.getElementById("output").innerHTML =
    "Fibonacci Series: " + fibonacci(num);
}
function isPrime(n){
    if(n<2) return false;
    for(let i=2;i<=Math.sqrt(n);i++){
        if(n%i==0)
            return false;
    }
    return true;
}
function primeNumbers(num){
    let primes=[];
    for(let i=2;i<=num;i++){
        if(isPrime(i)){
            primes.push(i);
        }
    }
    return primes.join(", ");
}
function displayPrimeNumbers(){
    let num = parseInt(document.getElementById("inputNum").value);
    document.getElementById("output").innerHTML =
    "Prime Numbers: " + primeNumbers(num);
}
function isPalindrome(num){
    let str = num.toString();
    let reversed = str.split("").reverse().join("");
    return str === reversed;
}
function checkPalindrome(){
    let num = parseInt(document.getElementById("inputNum").value);

    let result = isPalindrome(num) ? "Palindrome" : "Not a Palindrome";

    document.getElementById("output").innerHTML =
    "Palindrome Check: " + result;
}

<button onclick="displayPrimeNumbers()">Prime Numbers</button>
<button onclick="checkPalindrome()">Palindrome</button>
